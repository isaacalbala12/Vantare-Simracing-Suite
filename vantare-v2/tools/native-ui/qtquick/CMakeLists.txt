cmake_minimum_required(VERSION 3.22)
project(vantare_native_go_trial LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui Network Quick QuickControls2)
qt_standard_project_setup(REQUIRES 6.10)
qt_add_executable(vantare-native-go-qt main.cpp)
qt_add_qml_module(vantare-native-go-qt URI Vantare.NativeGoTrial VERSION 1.0 QML_FILES Main.qml)
target_link_libraries(vantare-native-go-qt PRIVATE Qt6::Core Qt6::Gui Qt6::Network Qt6::Quick Qt6::QuickControls2)
if(WIN32)
  target_link_libraries(vantare-native-go-qt PRIVATE user32)
endif()
